# `print` at the cursor and the three `print` overloads the templates use.
# The formatted ones are checked against C `snprintf` over 80,000 cases in
# `tests/fmt_parity.rs`; this is the on-screen check that the formatted
# text lands where the template put the cursor.
rot R0
font p17
pos top
clear
str 0 0 "at the cursor:"
cursor 0 16
print "12345"
printf1 0 32 93.5
printf0 0 48 12.0
printf1 0 60 -7.25
printi 0 16 42
printi 0 16 -7
# The degree sign, both ways the firmware writes it: `printc 176` -- what
# `print(static_cast<char>(176))` compiles to -- and a raw 0xB0 byte inside a
# printed string. `fmt::Formatted` builds the same thing as UTF-8 and hands the
# string to `print`, so this is the oracle's answer for the four temperature
# fields that used to draw nothing at all.
cursor 0 44
printc 176
printc 67
str 80 0 "\xb0C"
